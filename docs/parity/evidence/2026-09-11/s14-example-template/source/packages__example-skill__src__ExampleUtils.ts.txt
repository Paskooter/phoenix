import { skill } from '@jibo/interfaces';


/**
 * Used by nodes to create actions
 */
export class ExampleUtils {

    public static createSlimAction(text: string): skill.action.JCPAction {
        return {
            type: skill.action.ActionType.JCP,
            config: {
                version: '2.0',
                jcp: {
                    type: 'SLIM',
                    id: 'FAKE_SLIM_ID',
                    config: {
                        play: {
                            type: 'PLAY',
                            id: 'FAKE_PLAY_ID',
                            esml: text
                        },
                        listen: {
                            type: 'LISTEN',
                            id: 'FAKE_LISTEN_ID',
                            contexts: []
                        }
                    }
                }
            }
        };
    }
}
