#!/usr/bin/env node

require('@jibo/utils/common/run-service')('ExampleSkill', () => {
    const baseskill = require('@jibo/baseskill');
    const main = require('..');
    const path = require('path');
    const argv = require('minimist')(process.argv.slice(2));

    const DEFAULT_PORT = '8080';

    function printHelp() {
        console.log(`Usage: ${path.basename(process.argv[1])} [options]`);
        console.log(`  Options:`);
        console.log(`  --port, -p: [default: ${DEFAULT_PORT}] Port of service`);
    }

    const help = argv['h'] || argv['help'];
    const port = parseInt(argv['p'] || argv['port'] || process.env.ETCO_server_port || DEFAULT_PORT);

    if (help) {
        printHelp();
        return;
    }

    const exampleSkill = new main.ExampleSkill();

    const service = new baseskill.SkillService(exampleSkill, { debug: true });
    return service.init(port);
});
